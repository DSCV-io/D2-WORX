// DERIVED-VARIANT PROOF (toggled in by the runner; .txt so it's excluded from
// the default header-variant compile). Proves the DERIVED key source
// behaviorally: the gate dedupes on a CONTENT hash of (customerId, amountCents),
// so two requests with the SAME content dedupe even when their idempotencyKey
// FIELD values differ — and two requests with DIFFERENT content do NOT dedupe.
// Run against a lib generated from idempotent-derived.tsp.
//
// To exercise: regenerate from idempotent-derived.tsp, rename the header
// IdempotencyGateTests.cs out (it asserts the header-only "idempotency_key_missing"
// path, N/A for derived), drop this file in as .cs, build + test.

using System;
using System.Threading.Tasks;
using D2.Spike.Idempotency.Generated;
using Xunit;

namespace Spike.Idempotency.Test;

public sealed class DerivedKeyIdempotencyGateTests
{
    private static readonly TimeSpan s_ttl = TimeSpan.FromSeconds(300);

    private static (SubmitOrderIdempotencyGate Gate, CountingSubmitOrderHandler Handler, FakeClock Clock, InMemoryIdempotencyStore Store) NewSut()
    {
        var clock = new FakeClock();
        var store = new InMemoryIdempotencyStore(clock);
        var handler = new CountingSubmitOrderHandler();
        var gate = new SubmitOrderIdempotencyGate(handler, store); // C4: injected
        return (gate, handler, clock, store);
    }

    private static SubmitOrderInput Order(string key, string customer, int amount) =>
        new() { IdempotencyKey = key, CustomerId = customer, AmountCents = amount };

    // ---- C1 (derived): same CONTENT dedupes regardless of the key field -----
    [Fact]
    public async Task C1_DerivedKey_SameContentDifferentKeyField_HandlerRunsOnce()
    {
        var (gate, handler, _, _) = NewSut();

        // DIFFERENT idempotencyKey field values, but SAME hashed content.
        var a = await gate.ExecuteAsync(Order("key-X", customer: "cust-9", amount: 4200));
        var b = await gate.ExecuteAsync(Order("key-Y", customer: "cust-9", amount: 4200));

        // Deduped on content -> handler ran ONCE, second replayed.
        Assert.Equal(1, handler.Invocations);
        Assert.Equal("order-1", a.Value!.OrderId);
        Assert.Equal("order-1", b.Value!.OrderId);
    }

    [Fact]
    public async Task C1_DerivedKey_DifferentContent_HandlerRunsPerContent()
    {
        var (gate, handler, _, _) = NewSut();

        await gate.ExecuteAsync(Order("k", customer: "cust-9", amount: 4200));
        await gate.ExecuteAsync(Order("k", customer: "cust-9", amount: 9900)); // diff amount

        Assert.Equal(2, handler.Invocations); // distinct content -> not deduped
    }

    // ---- C2 (derived): ttl still respected on the content key ----------------
    [Fact]
    public async Task C2_DerivedKey_ReplayAfterTtl_ReInvokes()
    {
        var (gate, handler, clock, _) = NewSut();
        var input = Order("k", customer: "cust-9", amount: 4200);

        await gate.ExecuteAsync(input);
        Assert.Equal(1, handler.Invocations);

        clock.Advance(s_ttl + TimeSpan.FromSeconds(1));
        await gate.ExecuteAsync(input);
        Assert.Equal(2, handler.Invocations);
    }

    // ---- C4 (derived): store seam still injected -----------------------------
    [Fact]
    public async Task C4_DerivedKey_StoreSeamInjected()
    {
        var (gate, _, _, store) = NewSut();
        Assert.Equal(0, store.Count);
        await gate.ExecuteAsync(Order("k", customer: "cust-9", amount: 4200));
        Assert.Equal(1, store.Count);

        var ctors = typeof(SubmitOrderIdempotencyGate).GetConstructors();
        var only = Assert.Single(ctors);
        Assert.Equal(2, only.GetParameters().Length);
    }
}
